# 풀이봄 학생 앱 — 디자인·모션 코드 부록

> `디자인-모션-프롬프트.md` 와 **같이** 다른 AI 에게 준다. 아래는 지금 앱에 들어간 **실제 코드**를 그대로 옮긴 것이라, 이 코드를 그대로 쓰거나 같은 값으로 구현하게 하면 디자인·움직임이 똑같이 나온다.

```swift
import SwiftUI

// 앱 공통 모양: iOS 설정 앱처럼 둥근 묶음(FormGroup) 안에 한 줄씩(FormRow)

struct FormGroup<Content: View>: View {
    @ViewBuilder let content: Content

    var body: some View {
        VStack(spacing: 0) { content }
            .background(Color.paper, in: RoundedRectangle(cornerRadius: 20))
            .overlay(RoundedRectangle(cornerRadius: 20).stroke(Color.line, lineWidth: 1))
    }
}

struct FormRow<Content: View>: View {
    let label: String
    var last = false
    @ViewBuilder let content: Content

    var body: some View {
        VStack(spacing: 0) {
            HStack(spacing: 16) {
                Text(label)
                    .foregroundStyle(Color.ink)
                    .frame(width: 84, alignment: .leading)
                content
                    .font(.title3)
                    .foregroundStyle(Color.ink)
                    .tint(Color.honeyDeep)
                    .frame(maxWidth: .infinity, alignment: .trailing)
            }
            .frame(minHeight: 56)
            .padding(.horizontal, 18)
            if !last {
                Rectangle().fill(Color.line).frame(height: 1).padding(.leading, 18)
            }
        }
    }
}

// 고르기 칸 오른쪽 값 + ⌄
struct MenuValue: View {
    let text: String

    var body: some View {
        HStack(spacing: 6) {
            Text(text).foregroundStyle(Color.ink).lineLimit(1)
            Image(systemName: "chevron.up.chevron.down")
                .font(.caption.weight(.semibold))
                .foregroundStyle(Color.muted)
        }
    }
}

// 병아리 둥실둥실: 위아래로 떠오르며 좌우로 살짝 기운다
struct Bobbing: ViewModifier {
    @State private var up = false
    @Environment(\.accessibilityReduceMotion) private var reduceMotion

    func body(content: Content) -> some View {
        content
            .offset(y: up ? -10 : 0)
            .rotationEffect(.degrees(up ? -4 : 3), anchor: .bottom)
            .onAppear {
                guard !reduceMotion else { return }
                withAnimation(.easeInOut(duration: 1.5).repeatForever(autoreverses: true)) { up = true }
            }
    }
}

// 처음 나타날 때 아래에서 올라오며 (delay 로 차례차례)
struct RiseIn: ViewModifier {
    var delay: Double = 0
    @State private var shown = false

    func body(content: Content) -> some View {
        content
            .opacity(shown ? 1 : 0)
            .offset(y: shown ? 0 : 18)
            .onAppear {
                withAnimation(.spring(response: 0.5, dampingFraction: 0.8).delay(delay)) { shown = true }
            }
    }
}

extension View {
    // 시작 화면 병아리 → 로그인 화면 병아리로 이어서 움직이기
    @ViewBuilder func matchedLogo(_ ns: Namespace.ID?, isSource: Bool) -> some View {
        if let ns { matchedGeometryEffect(id: "logo-chick", in: ns, isSource: isSource) } else { self }
    }

    func bobbing() -> some View { modifier(Bobbing()) }
    func riseIn(delay: Double = 0) -> some View { modifier(RiseIn(delay: delay)) }
}

// 누르면 바로 살짝 들어가고, 떼면 탄력 있게 돌아오는 버튼 모양
struct PressableStyle: ButtonStyle {
    var scale: CGFloat = 0.96

    func makeBody(configuration: Configuration) -> some View {
        configuration.label
            .scaleEffect(configuration.isPressed ? scale : 1)
            .opacity(configuration.isPressed ? 0.85 : 1)
            .animation(
                configuration.isPressed ? .easeOut(duration: 0.08) : .spring(response: 0.32, dampingFraction: 0.55),
                value: configuration.isPressed
            )
    }
}

extension ButtonStyle where Self == PressableStyle {
    static var pressable: PressableStyle { PressableStyle() }
}

struct PrimaryButton: View {
    let title: String
    var busy = false
    var disabled = false
    let action: () -> Void

    var body: some View {
        Button(action: action) {
            HStack(spacing: 8) {
                if busy { ProgressView().tint(Color.ink) }
                Text(title).font(.headline)
            }
            .foregroundStyle(Color.ink)
            .frame(maxWidth: .infinity, minHeight: 56)
            .background(Color.honey, in: RoundedRectangle(cornerRadius: 18))
            .contentShape(RoundedRectangle(cornerRadius: 18))
        }
        .buttonStyle(.pressable)
        .disabled(busy || disabled)
        .opacity(busy || disabled ? 0.5 : 1)
    }
}

struct Notice: View {
    enum Kind { case error, ok }
    let text: String
    var kind: Kind = .error

            .background(kind == .error ? Color.salmonSoft : Color.sageSoft, in: RoundedRectangle(cornerRadius: 14))

}
```

## 3. 시작 화면: 병아리 점프 (Swift)

`ios/Pulibom/SplashView.swift`

```swift
import SwiftUI

// 앱을 켜면 잠깐 나오는 시작 화면: 병아리가 떨어지듯 나타나 두 번 통통 뛰고, 풀이봄 로고가 올라온다.
// 끝나면 병아리는 그대로 로그인 화면의 병아리 자리로 날아간다 (matchedGeometryEffect "logo-chick")
struct SplashView: View {
    let logo: Namespace.ID
    let onFinish: () -> Void // 점프·로고가 다 보이고 나면 (화면에 실제로 나온 때부터 센다)

    @State private var landed = false
    @State private var hop = 0
    @State private var showText = false

    var body: some View {
        ZStack {
            // 뒤의 로그인 화면이 비치지 않게 불투명한 바탕 위에 그라데이션
            Color.cream.ignoresSafeArea()
            LinearGradient(colors: [Color.honeySoft, Color.honey.opacity(0.55)], startPoint: .top, endPoint: .bottom)
                .ignoresSafeArea()

            VStack(spacing: 22) {
                Image("Chick")
                    .resizable()
                    .frame(width: 190, height: 190)
                    .matchedGeometryEffect(id: "logo-chick", in: logo)
                    .keyframeAnimator(initialValue: HopFrame(), trigger: hop) { content, f in
                        content
                            .scaleEffect(x: f.sx, y: f.sy, anchor: .bottom)
                            .offset(y: f.y)
                    } keyframes: { _ in
                        // 웅크림 → 높이 점프 → 착지 찌그러짐 → 작은 점프 → 제자리
                        KeyframeTrack(\.y) {
                            CubicKeyframe(0, duration: 0.12)
                            CubicKeyframe(-80, duration: 0.26)
                            CubicKeyframe(0, duration: 0.22)
                            CubicKeyframe(-34, duration: 0.18)
                            CubicKeyframe(0, duration: 0.16)
                        }
                        KeyframeTrack(\.sx) {
                            CubicKeyframe(1.16, duration: 0.12)
                            CubicKeyframe(0.9, duration: 0.26)
                            CubicKeyframe(1.18, duration: 0.22)
                            CubicKeyframe(0.96, duration: 0.18)
                            SpringKeyframe(1, duration: 0.3)
                        }
                        KeyframeTrack(\.sy) {
                            CubicKeyframe(0.84, duration: 0.12)
                            CubicKeyframe(1.12, duration: 0.26)
                            CubicKeyframe(0.82, duration: 0.22)
                            CubicKeyframe(1.05, duration: 0.18)
                            SpringKeyframe(1, duration: 0.3)
                        }
                    }
                    .shadow(color: Color.honeyDeep.opacity(0.35), radius: 18, y: 12)
                    .scaleEffect(landed ? 1 : 0.4)
                    .offset(y: landed ? 0 : -160)
                    .opacity(landed ? 1 : 0)

                VStack(spacing: 8) {
                    (Text("풀이봄") + Text(".").foregroundStyle(Color.honeyDeep))
                        .font(.system(size: 52, weight: .heavy))
                        .foregroundStyle(Color.ink)
                    Text("풀이를 모아, 배움을 잇다")
                        .font(.title3)
                        .foregroundStyle(Color.olive)
                }
                .opacity(showText ? 1 : 0)
                .offset(y: showText ? 0 : 16)
            }
        }
        .task {
            withAnimation(.spring(response: 0.45, dampingFraction: 0.62)) { landed = true }
            try? await Task.sleep(for: .seconds(0.45))
            hop += 1
            withAnimation(.easeOut(duration: 0.5).delay(0.25)) { showText = true }
            try? await Task.sleep(for: .seconds(1.7))
            onFinish()
        }
    }
}

private struct HopFrame {
    var y: CGFloat = 0
    var sx: CGFloat = 1
    var sy: CGFloat = 1
}
```

```swift
import SwiftUI

struct ContentView: View {
    // 풀이봄 웹 서버 주소 (예: http://192.168.0.10:3000). 처음 켤 때 입력받는다.
    @AppStorage("serverURL") private var serverURL = ""
    @State private var showSettings = false
    @State private var loadError: String?
    @State private var reloadToken = 0
    // 앱을 켤 때마다 로그인부터 (nil = 로그인 화면)
    @State private var student: LoggedInStudent?
    // 시작 화면 (켤 때 한 번). 병아리가 로그인 화면 자리로 이어서 날아가도록 같은 이름공간을 쓴다
    @State private var splash = true
    @Namespace private var logo

    var body: some View {
        ZStack {
            Color.cream.ignoresSafeArea()

            if let url = ServerAddress.url(from: serverURL), student == nil {
                LoginView(
                    server: url,
                    onLogin: { s in withAnimation(.easeInOut(duration: 0.35)) { student = s } },
                    onOpenSettings: { showSettings = true },
                    logo: logo,
                    splashing: splash
                )
                .id(serverURL)
                .transition(.opacity.combined(with: .scale(scale: 0.97)))
            } else if let url = ServerAddress.url(from: serverURL), let student {
                StudentShellView(
                    url: url,
                    student: student,
                    reloadToken: reloadToken,
                    onLoaded: { loadError = nil },
                    onError: { loadError = $0 },
                    onLoggedOut: { withAnimation(.easeInOut(duration: 0.35)) { self.student = nil } },
                    onOpenSettings: { showSettings = true }
                )
                .id(serverURL)
                .transition(.opacity.combined(with: .move(edge: .trailing)))

                if let loadError {
                    ConnectionErrorView(
                        address: serverURL,
                        message: loadError,
                        onRetry: {
                            self.loadError = nil
                            reloadToken += 1
                        },
                        onChangeServer: { showSettings = true }
                    )
                }
            } else {
                ServerSettingsView(serverURL: $serverURL, isFirstRun: true)
            }

            if splash {
                // 병아리가 두 번 뛰고 나면 로그인 화면으로 (병아리는 그대로 로그인 화면 자리로 날아감)
                SplashView(logo: logo) {
                    withAnimation(.spring(response: 0.75, dampingFraction: 0.82)) { splash = false }
                }
                .transition(.opacity)
                .zIndex(1)
            }
        }
        .sheet(isPresented: $showSettings) {
            ServerSettingsView(serverURL: $serverURL, isFirstRun: false)
        }
        .onChange(of: serverURL) {
            loadError = nil
            student = nil
        }
        .onAppear {
            // 수업 중에 화면이 꺼지지 않게
            UIApplication.shared.isIdleTimerDisabled = true
        }
    }
}

struct ConnectionErrorView: View {
    let address: String
    let message: String
    let onRetry: () -> Void
    let onChangeServer: () -> Void

    var body: some View {
        ZStack {
            Color.cream.ignoresSafeArea()
            VStack(spacing: 16) {
                Text("🐣").font(.system(size: 64))
                Text("풀이봄 서버에 연결할 수 없어요")
                    .font(.title2.bold())
                    .foregroundStyle(Color.ink)
                Text(address)
                    .font(.callout.monospaced())
                    .foregroundStyle(Color.muted)
                Text(message)
                    .font(.footnote)
                    .foregroundStyle(Color.muted)
                    .multilineTextAlignment(.center)
                Text("맥과 아이패드가 같은 와이파이에 있는지, 맥에서 서버(npm run dev)가 켜져 있는지 확인해 주세요.")
                    .font(.footnote)
                    .foregroundStyle(Color.muted)
                    .multilineTextAlignment(.center)
                HStack(spacing: 12) {
                    Button("서버 주소 바꾸기", action: onChangeServer)
                        .buttonStyle(.bordered)
                    Button("다시 시도", action: onRetry)
                        .buttonStyle(.borderedProminent)
                }
                .tint(Color.ink)
                .padding(.top, 8)
            }
            .frame(maxWidth: 440)
            .padding(32)
        }
    }
}
```

## 5. 로그인 화면 (Swift)

`ios/Pulibom/LoginView.swift`

```swift
import SwiftUI

// 앱을 켜면 나오는 학생 로그인
// 가로: 왼쪽 병아리·환영 문구 / 오른쪽 입력. 세로: 위아래로
// 아이패드 한 대를 여러 학생이 돌려 써도 누가 푸는지 남도록, 앱을 켤 때마다 새로 로그인한다
struct LoginView: View {
    let server: URL
    let onLogin: (LoggedInStudent) -> Void
    let onOpenSettings: () -> Void
    var logo: Namespace.ID? = nil // 시작 화면 병아리와 이어지게
    var splashing = false

    @State private var students: [Student] = []
    @State private var loading = true
    @State private var loadError: String?
    @State private var classId = ""
    @State private var studentId = ""
    @State private var password = ""
    @State private var error: String?
    @State private var busy = false
    @State private var showSignup = false
    @FocusState private var passwordFocused: Bool

    // 웹 로그인과 같이: 반을 먼저 고르고 → 그 반 학생 중에서 번호를 고른다
    private var classes: [(id: String, name: String)] {
        var seen = Set<String>()
        return students.filter { seen.insert($0.classId).inserted }.map { ($0.classId, $0.className) }
    }

    private var inClass: [Student] { students.filter { $0.classId == classId } }
    private var selected: Student? { students.first { $0.id == studentId } }

    var body: some View {
        GeometryReader { geo in
            let wide = geo.size.width > geo.size.height
            // 세로 화면에서 비밀번호를 칠 땐 병아리 영역을 줄여서 키보드에 칸이 가리지 않게
            let compact = !wide && passwordFocused
            ZStack(alignment: .topTrailing) {
                Color.cream.ignoresSafeArea()

                if wide {
                    HStack(spacing: 0) {
                        hero(big: true)
                            .frame(width: geo.size.width * 0.44)
                        formSide
                            .frame(maxWidth: .infinity)
                    }
                } else {
                    VStack(spacing: 0) {
                        hero(big: false)
                            .frame(height: compact ? 0 : geo.size.height * 0.34)
                            .clipped()
                            .opacity(compact ? 0 : 1)
                        formSide
                    }
                    .animation(.easeInOut(duration: 0.25), value: compact)
                }

                Button(action: onOpenSettings) {
                    Image(systemName: "gearshape")
                        .font(.title3)
                        .foregroundStyle(Color.muted)
                        .frame(width: 44, height: 44)
                }
                .padding(12)
                .accessibilityLabel("서버 주소 설정")
            }
        }
        .task { await load() }
        .sheet(isPresented: $showSignup) {
            SignupView(server: server, onSignup: { s in
                showSignup = false
                onLogin(s)
            })
        }
    }

    // 병아리 + 환영 문구
    private func hero(big: Bool) -> some View {
        ZStack {
            LinearGradient(colors: [Color.honeySoft, Color.honey.opacity(0.55)], startPoint: .top, endPoint: .bottom)
                .ignoresSafeArea()
            VStack(spacing: big ? 18 : 10) {
                Image("Chick")
                    .resizable()
                    .frame(width: big ? 180 : 110, height: big ? 180 : 110)
                    .matchedLogo(logo, isSource: !splashing)
                    .shadow(color: Color.honeyDeep.opacity(0.35), radius: 16, y: 10)
                    .bobbing()
                (Text("풀이봄") + Text(".").foregroundStyle(Color.honeyDeep))
                    .font(.system(size: big ? 44 : 32, weight: .heavy))
                    .foregroundStyle(Color.ink)
                Text("풀이를 모아, 배움을 잇다")
                    .font(big ? .title3 : .body)
                    .foregroundStyle(Color.olive)
            }
            .padding(24)
        }
    }

    private var formSide: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 24) {
                VStack(alignment: .leading, spacing: 6) {
                    Text("로그인")
                        .font(.largeTitle.bold())
                        .foregroundStyle(Color.ink)
                    Text("반과 번호를 고르고 비밀번호를 넣어 주세요.")
                        .foregroundStyle(Color.muted)
                }
                .riseIn()

                if loading {
                    HStack(spacing: 10) {
                        ProgressView()
                        Text("학생 명단 불러오는 중…").foregroundStyle(Color.muted)
                    }
                    .frame(maxWidth: .infinity, minHeight: 180)
                } else if let loadError {
                    loadErrorView(loadError)
                } else {
                    form.riseIn(delay: 0.08)
                }
            }
            .frame(maxWidth: 440)
            .padding(.horizontal, 32)
            .padding(.vertical, 56)
            .frame(maxWidth: .infinity)
        }
        .scrollDismissesKeyboard(.interactively)
    }

    private var form: some View {
        VStack(alignment: .leading, spacing: 16) {
            FormGroup {
                FormRow(label: "반") {
                    Menu {
                        Picker("반", selection: Binding(get: { classId }, set: { pickClass($0) })) {
                            ForEach(classes, id: \.id) { Text($0.name).tag($0.id) }
                        }
                    } label: {
                        MenuValue(text: classes.first { $0.id == classId }?.name ?? "선택")
                    }
                }
                FormRow(label: "번호") {
                    Menu {
                        Picker("번호", selection: $studentId) {
                            ForEach(inClass) { Text(label(for: $0)).tag($0.id) }
                        }
                    } label: {
                        MenuValue(text: selected.map(label(for:)) ?? "선택")
                    }
                }
                FormRow(label: "비밀번호", last: true) {
                    SecureField("숫자", text: $password)
                        .multilineTextAlignment(.trailing)
                        .keyboardType(.numberPad)
                        .textContentType(.password)
                        .focused($passwordFocused)
                        .onChange(of: password) {
                            error = nil
                            let cleaned = String(password.filter(\.isNumber).prefix(8))
                            if cleaned != password { password = cleaned }
                        }
                        .onSubmit { Task { await login() } }
                }
            }
            .onChange(of: studentId) { error = nil }

            if let error { Notice(text: error) }

            PrimaryButton(title: busy ? "들어가는 중…" : "과제 풀러 가기", busy: busy, disabled: password.isEmpty) {
                Task { await login() }
            }

            HStack(spacing: 4) {
                Text("처음 왔나요?").foregroundStyle(Color.muted)
                Button("반 코드로 회원가입") { showSignup = true }
                    .fontWeight(.semibold)
                    .foregroundStyle(Color.honeyDeep)
            }
            .frame(maxWidth: .infinity)
            .padding(.top, 4)

            VStack(spacing: 6) {
                Button("로그인 건너뛰고 둘러보기") { Task { await login(skip: true) } }
                    .font(.footnote)
```

## 7. 메시지 위젯·말풍선 (Swift)

```swift
import SwiftUI

// 선생님과 메시지: 상단 바 💬 를 누르면 그 아래로 뜨는 작은 창 (StudentShellView 의 chatPanel)
// 열려 있는 동안 4초마다 새 메시지를 가져온다. 선생님 쪽은 웹 학생 상세의 '메시지' 탭
struct ChatWidget: View {
    let server: URL
    var onClose: (() -> Void)? = nil

    @State private var messages: [ChatMessage] = []
    @State private var loaded = false
    @State private var text = ""
    @State private var sending = false
    @State private var error: String?
    @FocusState private var focused: Bool

    var body: some View {
        VStack(spacing: 0) {
            header
            Rectangle().fill(Color.line).frame(height: 1)
            list
            Rectangle().fill(Color.line).frame(height: 1)
            input
        }
        .background(Color.cream)
        .task {
            // 열려 있는 동안 계속 새로 고침 (창을 닫으면 task 가 끝난다)
            while !Task.isCancelled {
                await refresh()
                try? await Task.sleep(for: .seconds(4))
            }
        }
    }

    private var header: some View {
        HStack(spacing: 10) {
            Image("Chick").resizable().frame(width: 30, height: 30)
            VStack(alignment: .leading, spacing: 1) {
                Text("선생님께 메시지").font(.headline).foregroundStyle(Color.ink)
                Text("궁금한 점을 편하게 물어보세요").font(.caption).foregroundStyle(Color.muted)
            }
            Spacer()
            if let onClose {
                Button(action: onClose) {
                    Image(systemName: "xmark")
                        .font(.system(size: 13, weight: .bold))
                        .foregroundStyle(Color.muted)
                        .frame(width: 30, height: 30)
                        .background(Color.sand, in: Circle())
                }
                .buttonStyle(.pressable)
                .accessibilityLabel("닫기")
            }
        }
        .padding(.horizontal, 16)
        .padding(.vertical, 12)
        .background(Color.paper)
    }

    private var list: some View {
        ScrollViewReader { proxy in
            ScrollView {
                LazyVStack(spacing: 10) {
                    if loaded && messages.isEmpty && error == nil {
                        VStack(spacing: 8) {
                            Image("Chick").resizable().frame(width: 56, height: 56).bobbing()
                            Text("아직 주고받은 메시지가 없어요.\n모르는 문제나 궁금한 점을 보내 보세요!")
                                .font(.footnote)
                                .multilineTextAlignment(.center)
                                .foregroundStyle(Color.muted)
                        }
                        .padding(.top, 60)
                    }
                    ForEach(messages) { m in
                        bubble(m).id(m.id)
                            .transition(.asymmetric(insertion: .scale(scale: 0.85, anchor: m.mine ? .bottomTrailing : .bottomLeading).combined(with: .opacity), removal: .opacity))
                    }
                    if let error { Notice(text: error).padding(.top, 8) }
                }
                .padding(14)
            }
            .onChange(of: messages.last?.id) {
                if let last = messages.last?.id { withAnimation { proxy.scrollTo(last, anchor: .bottom) } }
            }
            // 키보드가 올라와 창이 줄어들어도 마지막 메시지가 보이게
            .onChange(of: focused) {
                guard focused, let last = messages.last?.id else { return }
                Task {
                    try? await Task.sleep(for: .seconds(0.3))
                    withAnimation { proxy.scrollTo(last, anchor: .bottom) }
                }
            }
            .scrollDismissesKeyboard(.interactively)
            .onAppear {
                if let last = messages.last?.id { proxy.scrollTo(last, anchor: .bottom) }
            }
        }
    }

    private func bubble(_ m: ChatMessage) -> some View {
        HStack {
            if m.mine { Spacer(minLength: 48) }
            VStack(alignment: m.mine ? .trailing : .leading, spacing: 3) {
                if !m.mine {
                    Text(m.senderName).font(.caption2).foregroundStyle(Color.muted).padding(.horizontal, 4)
                }
                Text(m.body)
                    .font(.subheadline)
                    .foregroundStyle(Color.ink)
                    .padding(.horizontal, 13)
                    .padding(.vertical, 9)
                    .background(m.mine ? Color.honey : Color.paper, in: BubbleShape(mine: m.mine))
                    .overlay { if !m.mine { BubbleShape(mine: false).stroke(Color.line, lineWidth: 1) } }
                Text(time(m.createdAt) + (m.mine && m.readAt != nil ? " · 읽음" : ""))
                    .font(.caption2)
                    .foregroundStyle(Color.muted)
                    .padding(.horizontal, 4)
            }
            if !m.mine { Spacer(minLength: 48) }
        }
    }

    private var input: some View {
        HStack(spacing: 8) {
            TextField("선생님께 보낼 말", text: $text, axis: .vertical)
                .lineLimit(1...4)
                .font(.body)
                .foregroundStyle(Color.ink)
                .tint(Color.honeyDeep)
                .focused($focused)
                .padding(.horizontal, 14)
                .padding(.vertical, 10)
                .background(Color.white, in: RoundedRectangle(cornerRadius: 20))
                .overlay(RoundedRectangle(cornerRadius: 20).stroke(Color.line, lineWidth: 1))
                .onSubmit { Task { await send() } }
            Button {
                Task { await send() }
            } label: {
                Image(systemName: sending ? "ellipsis.circle.fill" : "arrow.up.circle.fill")
                    .font(.system(size: 34))
                    .foregroundStyle(canSend ? Color.honeyDeep : Color.line)
                    .symbolEffect(.pulse, isActive: sending)
            }
            .buttonStyle(.pressable)
            .disabled(!canSend)
            .accessibilityLabel("보내기")
        }
        .padding(12)
        .background(Color.paper)
    }

    private var canSend: Bool { !sending && !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty }

    private func refresh() async {
        do {
            let next = try await PulibomAPI.messages(server)
            if next != messages { withAnimation(.spring(response: 0.35, dampingFraction: 0.8)) { messages = next } }
            if error != nil { error = nil }
        } catch {
            self.error = error.localizedDescription.contains("로그인") ? "둘러보기 중에는 메시지를 쓸 수 없어요. 내 번호로 로그인해 주세요." : error.localizedDescription
        }
        loaded = true
    }

    private func send() async {
        let body = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !body.isEmpty, !sending else { return }
        sending = true
        defer { sending = false }
        do {
            let m = try await PulibomAPI.sendMessage(server, body: body)
            UIImpactFeedbackGenerator(style: .light).impactOccurred()
            withAnimation(.spring(response: 0.35, dampingFraction: 0.8)) { messages.append(m) }
            text = ""
            error = nil
        } catch {
            self.error = error.localizedDescription
        }
    }

        let f = DateFormatter()
        f.locale = Locale(identifier: "ko_KR")
        f.dateFormat = Calendar.current.isDateInToday(Date(timeIntervalSince1970: ms / 1000)) ? "a h:mm" : "M/d a h:mm"
        return f.string(from: Date(timeIntervalSince1970: ms / 1000))
    }
```

```tsx
import { useEffect, useRef, useState, type ReactNode } from "react";

// 학생 과제 목록 탭 (아이패드 앱 하단 탭 바와 연결)
// 세 탭 내용을 미리 다 받아 두고 화면에서만 바꿔서, 서버를 다시 부르지 않고 바로 부드럽게 넘어간다.
// 앱은 window 이벤트 'pulibom:tab' 으로 탭을 바꾸고, 주소(?tab=)도 같이 바꿔서 정렬·새로고침 때 그대로 남는다.
export const TAB_ORDER = ["todo", "done", "past"] as const;
export type StudentTab = (typeof TAB_ORDER)[number];

export default function StudentTabs({ initial, panels }: { initial: StudentTab; panels: Record<StudentTab, ReactNode> }) {
  const [active, setActive] = useState(initial);
  const [dir, setDir] = useState<"left" | "right" | null>(null);
  const activeRef = useRef(initial);

  useEffect(() => {
    const go = (next: StudentTab) => {
      if (next === activeRef.current) return;
      setDir(TAB_ORDER.indexOf(next) > TAB_ORDER.indexOf(activeRef.current) ? "right" : "left");
      activeRef.current = next;
      setActive(next);
      const url = new URL(location.href);
      url.searchParams.set("tab", next);
      history.replaceState(history.state, "", url);
      window.scrollTo({ top: 0 });
    };
    const onTab = (e: Event) => {
      const t = (e as CustomEvent<string>).detail;
      if (TAB_ORDER.includes(t as StudentTab)) go(t as StudentTab);
    };
    window.addEventListener("pulibom:tab", onTab);
    const w = window as unknown as { __pulibomTabs?: boolean; __pulibomReady?: boolean };
    w.__pulibomTabs = true;
    w.__pulibomReady = true; // 앱이 Next 라우터로 화면을 옮겨도 되는 때
    return () => {
      window.removeEventListener("pulibom:tab", onTab);
      w.__pulibomTabs = false;
    };
  }, []);

  return (
    <div key={active} className={dir ? `tab-in-${dir}` : undefined}>
      {panels[active]}
    </div>
  );
}
```
